29Node *FirstOfNodeFactory::getNode(const QString &tagContent, Parser *p) const
30{
31 auto expr = smartSplit(tagContent);
32
33 const auto tagName = expr.takeAt(0);
34
35 if (expr.isEmpty()) {
36 throw Cutelee::Exception(
37 TagSyntaxError,
38 QStringLiteral("%1 expects at least one argument").arg(tagName));
39 }
40
41 return new FirstOfNode(getFilterExpressionList(expr, p), p);
42}
