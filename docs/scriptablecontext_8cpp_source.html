45QString ScriptableContext::render(const QList<QObject *> &list) const
46{
47 NodeList nodeList;
48 for (auto n : list) {
49 auto node = qobject_cast<Node *>(n);
50 if (node)
51 nodeList << node;
52 }
53 QString ret;
54 QTextStream t(&ret);
55 OutputStream stream(&t);
56 nodeList.render(&stream, m_c);
57 return ret;
